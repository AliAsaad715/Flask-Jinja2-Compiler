from flask import Flask, render_template, redirect, url_for

app = Flask(__name__)

products = [
    {
        'id': 1,
        'name': 'لابتوب ديل',
        'price': 2500.00,
        'image': '',
        'details': 'لابتوب ديل بمواصفات عالية، معالج i7، ذاكرة 16GB، مساحة تخزين 512GB SSD'
    },
    {
        'id': 2,
        'name': 'هاتف سامسونج',
        'price': 1800.00,
        'image': '',
        'details': 'هاتف سامسونج ذكي، شاشة 6.2 بوصة، كاميرا 48 ميجابكسل، بطارية 4000 مللي أمبير'
    }
]

@app.route('/')
def index():
    return redirect(url_for('products_list'))

@app.route('/products')
def products_list():
    return render_template('products.html', products=products)

@app.route('/product/<int:product_id>')
def product_detail(product_id):
    product = next((p for p in products if p['id'] == product_id), None)
    if product is None:
        return 'المنتج غير موجود', 404
    return render_template('product_detail.html', product=product)

@app.route('/add-product')
def add_product():
    return render_template(
        'add_product.html',
        generation_note='أضف المنتج إلى مصفوفة products ثم أعد تشغيل المولد.'
    )

@app.route('/delete-product/<int:product_id>')
def delete_product(product_id):
    product = next((p for p in products if p['id'] == product_id), None)
    if product is None:
        return 'المنتج غير موجود', 404
    return render_template('delete_product.html', product=product)

if __name__ == '__main__':
    app.run()
